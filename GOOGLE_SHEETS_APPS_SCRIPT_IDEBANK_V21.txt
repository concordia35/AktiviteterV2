GOOGLE APPS SCRIPT - IDÉBANK V2.1
=================================

Denne kode er lavet til den simple Idébank-model.

Den understøtter:
- Broderinitiativer
- Deltagere til initiativer
- Idébank
- Interesseafstemninger direkte i samme Idebank-ark

Der bruges kun én ny fane:

  Idebank

const SHEET_NAMES = {
  initiatives: 'Initiativer',
  participants: 'Deltagere',
  ideas: 'Idebank'
};

const HEADERS = {
  initiatives: ['ID', 'Status', 'Titel', 'Dato', 'Tidspunkt', 'Sted', 'Kontaktperson', 'Beskrivelse', 'Ikon'],
  participants: ['Tidspunkt', 'Aktivitet ID', 'Aktivitet', 'Navn'],
  ideas: [
    'ID',
    'Indsendt',
    'Navn',
    'Titel',
    'Beskrivelse',
    'Kategori',
    'Deltagerestimat',
    'Vil hjælpe',
    'Status',
    'Interesserede',
    'Interesserede navne',
    'Statusnote',
    'Dato for arrangement',
    'Tidspunkt',
    'Sted',
    'Kontaktperson'
  ]
};

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu('Aktivitetsapp')
    .addItem('Opsæt/kontrollér ark', 'setupActivitySheets')
    .addToUi();
}

function setupActivitySheets() {
  ensureSheet_(SHEET_NAMES.initiatives, HEADERS.initiatives);
  ensureSheet_(SHEET_NAMES.participants, HEADERS.participants);
  ensureSheet_(SHEET_NAMES.ideas, HEADERS.ideas);

  const ss = SpreadsheetApp.getActive();
  const ideaSheet = ss.getSheetByName(SHEET_NAMES.ideas);
  const statusCol = HEADERS.ideas.indexOf('Status') + 1;

  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(['Ny', 'Under behandling', 'Afstemning', 'Planlagt', 'Gennemført', 'Afvist'], true)
    .setAllowInvalid(false)
    .build();

  ideaSheet
    .getRange(2, statusCol, Math.max(ideaSheet.getMaxRows() - 1, 1), 1)
    .setDataValidation(rule);

  const interestedCol = HEADERS.ideas.indexOf('Interesserede') + 1;
  ideaSheet
    .getRange(2, interestedCol, Math.max(ideaSheet.getMaxRows() - 1, 1), 1)
    .setNumberFormat('0');

  SpreadsheetApp.getUi().alert('Ark er klar. Brug fanen Idebank til idéer.');
}

function doGet(e) {
  try {
    ensureSheet_(SHEET_NAMES.initiatives, HEADERS.initiatives);
    ensureSheet_(SHEET_NAMES.participants, HEADERS.participants);
    ensureSheet_(SHEET_NAMES.ideas, HEADERS.ideas);

    const p = e && e.parameter ? e.parameter : {};
    const action = String(p.action || 'list');

    if (action === 'list') {
      return jsonResponse({
        ok: true,
        initiatives: readObjects_(SHEET_NAMES.initiatives),
        participants: readObjects_(SHEET_NAMES.participants),
        ideas: readObjects_(SHEET_NAMES.ideas),
        ideaVotes: []
      });
    }

    if (action === 'getInitiatives') {
      return jsonResponse({ ok: true, initiatives: readObjects_(SHEET_NAMES.initiatives) });
    }

    if (action === 'getParticipants') {
      return jsonResponse({ ok: true, participants: readObjects_(SHEET_NAMES.participants) });
    }

    if (action === 'getIdeas') {
      return jsonResponse({ ok: true, ideas: readObjects_(SHEET_NAMES.ideas) });
    }

    if (action === 'getIdeaVotes') {
      return jsonResponse({ ok: true, ideaVotes: [] });
    }

    if (action === 'submitInitiative') return submitInitiative_(p);
    if (action === 'joinActivity') return joinActivity_(p);
    if (action === 'submitIdea') return submitIdea_(p);
    if (action === 'voteIdea') return voteIdea_(p);

    return jsonResponse({ ok: false, error: 'Ukendt action: ' + action });
  } catch (err) {
    return jsonResponse({ ok: false, error: String(err && err.message ? err.message : err) });
  }
}

function doPost(e) {
  return doGet(e);
}

function submitInitiative_(p) {
  const sheet = ensureSheet_(SHEET_NAMES.initiatives, HEADERS.initiatives);
  const title = clean_(p.title);
  const date = clean_(p.date);

  if (!title || !date) {
    return jsonResponse({ ok: false, error: 'Titel og dato mangler.' });
  }

  const id = 'init-' + Utilities.getUuid().slice(0, 8);

  appendObject_(sheet, {
    'ID': id,
    'Status': 'Ny',
    'Titel': title,
    'Dato': date,
    'Tidspunkt': clean_(p.time),
    'Sted': clean_(p.place),
    'Kontaktperson': clean_(p.host),
    'Beskrivelse': clean_(p.text),
    'Ikon': '🤝'
  });

  return jsonResponse({ ok: true, id: id });
}

function joinActivity_(p) {
  const sheet = ensureSheet_(SHEET_NAMES.participants, HEADERS.participants);
  const activityId = clean_(p.activityId);
  const name = clean_(p.name);

  if (!activityId || !name) {
    return jsonResponse({ ok: false, error: 'Aktivitet og navn mangler.' });
  }

  const rows = readObjects_(SHEET_NAMES.participants);
  const exists = rows.some(row => same_(row['Aktivitet ID'], activityId) && same_(row['Navn'], name));

  if (!exists) {
    appendObject_(sheet, {
      'Tidspunkt': new Date(),
      'Aktivitet ID': activityId,
      'Aktivitet': clean_(p.activity),
      'Navn': name
    });
  }

  return jsonResponse({ ok: true, duplicate: exists });
}

function submitIdea_(p) {
  const sheet = ensureSheet_(SHEET_NAMES.ideas, HEADERS.ideas);
  const title = clean_(p.title);
  const name = clean_(p.name);

  if (!title || !name) {
    return jsonResponse({ ok: false, error: 'Titel og navn mangler.' });
  }

  const id = 'idea-' + Utilities.getUuid().slice(0, 8);

  appendObject_(sheet, {
    'ID': id,
    'Indsendt': new Date(),
    'Navn': name,
    'Titel': title,
    'Beskrivelse': clean_(p.text),
    'Kategori': clean_(p.category) || 'Andet',
    'Deltagerestimat': clean_(p.estimate),
    'Vil hjælpe': clean_(p.help) || 'Nej',
    'Status': 'Ny',
    'Interesserede': 0,
    'Interesserede navne': '',
    'Statusnote': '',
    'Dato for arrangement': '',
    'Tidspunkt': '',
    'Sted': '',
    'Kontaktperson': ''
  });

  return jsonResponse({ ok: true, id: id });
}

function voteIdea_(p) {
  const ideaId = clean_(p.ideaId);
  const name = clean_(p.name);
  const vote = clean_(p.vote) || 'Ja';

  if (!ideaId || !name) {
    return jsonResponse({ ok: false, error: 'Forslag og navn mangler.' });
  }

  const sheet = ensureSheet_(SHEET_NAMES.ideas, HEADERS.ideas);
  const values = sheet.getDataRange().getValues();

  if (values.length < 2) {
    return jsonResponse({ ok: false, error: 'Ingen idéer fundet.' });
  }

  const headers = values[0].map(h => String(h || '').trim());
  const idCol = headers.indexOf('ID');
  const interestedCol = headers.indexOf('Interesserede');
  const namesCol = headers.indexOf('Interesserede navne');

  let targetRow = -1;

  for (let i = 1; i < values.length; i++) {
    if (same_(values[i][idCol], ideaId)) {
      targetRow = i + 1;
      break;
    }
  }

  if (targetRow < 2) {
    return jsonResponse({ ok: false, error: 'Idéen blev ikke fundet.' });
  }

  // For at holde arket simpelt tæller vi kun Ja-interesse.
  // Nej-svar gemmes ikke.
  if (vote.toLowerCase() !== 'ja') {
    return jsonResponse({ ok: true, counted: false });
  }

  const currentNames = splitNames_(sheet.getRange(targetRow, namesCol + 1).getValue());
  const exists = currentNames.some(n => same_(n, name));

  if (!exists) {
    currentNames.push(name);
  }

  sheet.getRange(targetRow, namesCol + 1).setValue(currentNames.join('; '));
  sheet.getRange(targetRow, interestedCol + 1).setValue(currentNames.length);

  return jsonResponse({ ok: true, duplicate: exists, count: currentNames.length });
}

function ensureSheet_(name, headers) {
  const ss = SpreadsheetApp.getActive();
  let sheet = ss.getSheetByName(name);

  if (!sheet) {
    sheet = ss.insertSheet(name);
  }

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
    sheet.setFrozenRows(1);
    sheet.autoResizeColumns(1, headers.length);
    return sheet;
  }

  const lastColumn = Math.max(sheet.getLastColumn(), headers.length);
  const current = sheet
    .getRange(1, 1, 1, lastColumn)
    .getValues()[0]
    .map(v => String(v || '').trim());

  headers.forEach(header => {
    if (!current.includes(header)) {
      sheet.getRange(1, sheet.getLastColumn() + 1).setValue(header);
    }
  });

  sheet.setFrozenRows(1);
  sheet.autoResizeColumns(1, sheet.getLastColumn());

  return sheet;
}

function appendObject_(sheet, obj) {
  const headers = sheet
    .getRange(1, 1, 1, sheet.getLastColumn())
    .getValues()[0]
    .map(h => String(h || '').trim());

  const row = headers.map(header => obj[header] !== undefined ? obj[header] : '');
  sheet.appendRow(row);
}

function readObjects_(sheetName) {
  const ss = SpreadsheetApp.getActive();
  const sheet = ss.getSheetByName(sheetName);

  if (!sheet || sheet.getLastRow() < 2) {
    return [];
  }

  const values = sheet.getDataRange().getValues();
  const headers = values.shift().map(h => String(h || '').trim());

  return values
    .filter(row => row.some(cell => String(cell || '').trim() !== ''))
    .map(row => {
      const obj = {};
      headers.forEach((header, i) => obj[header] = formatCell_(row[i]));
      return obj;
    });
}

function formatCell_(value) {
  if (value instanceof Date) {
    return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
  }
  return String(value == null ? '' : value).trim();
}

function splitNames_(value) {
  return String(value || '')
    .split(/[;,\n]/)
    .map(v => v.trim())
    .filter(Boolean);
}

function clean_(value) {
  return String(value == null ? '' : value).trim();
}

function same_(a, b) {
  return clean_(a).toLowerCase() === clean_(b).toLowerCase();
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
